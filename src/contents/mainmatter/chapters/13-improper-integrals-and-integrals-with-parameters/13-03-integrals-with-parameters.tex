\section{含参变量的积分}

\begin{problem}{含参量积分的极限}{Limit-of-Parametric-Integral}
    试用两种方法计算以下极限：
    \begin{enumerate}
        \item $\lim_{\alpha \to 0} \int_{-1}^{1} \sqrt{x^2 + \alpha^2} \d x$;
        \item $\lim_{\alpha \to 0} \int_{\alpha}^{1+\alpha} \frac{1}{1 + x^2 + \alpha^2} \d x$.
    \end{enumerate}
\end{problem}

\begin{solution}
    \ansitem{1}

    方法一：直接积分。
    因为被积函数为偶函数，有
    \begin{equation}
        \begin{aligned}
            \int_{-1}^{1} \sqrt{x^2 + \alpha^2} \d x & = 2 \int_{0}^{1} \sqrt{x^2 + \alpha^2} \d x                                                          \\
                                                     & = \left[ x\sqrt{x^2 + \alpha^2} + \alpha^2 \ln\left(x + \sqrt{x^2 + \alpha^2}\right) \right]_{0}^{1} \\
                                                     & = \sqrt{1 + \alpha^2} + \alpha^2 \ln\left(1 + \sqrt{1 + \alpha^2}\right) - \alpha^2 \ln|\alpha|.
        \end{aligned}
    \end{equation}
    由于 $\lim_{\alpha \to 0} \alpha^2 \ln|\alpha| = 0$，对上式取极限得
    \begin{equation}
        \boxed{\lim_{\alpha \to 0} \int_{-1}^{1} \sqrt{x^2 + \alpha^2} \d x = 1}
    \end{equation}

    方法二：利用含参量积分的性质。
    设 $f(x, \alpha) = \sqrt{x^2 + \alpha^2}$。在闭区域 $[-1, 1] \times [-1, 1]$ 上，$f(x, \alpha)$ 连续，因此可以将极限运算与积分运算交换顺序，即
    \begin{equation}
        \begin{aligned}
            \lim_{\alpha \to 0} \int_{-1}^{1} \sqrt{x^2 + \alpha^2} \d x & = \int_{-1}^{1} \lim_{\alpha \to 0} \sqrt{x^2 + \alpha^2} \d x \\
                                                                         & = \int_{-1}^{1} |x| \d x                                       \\
                                                                         & = 2 \int_{0}^{1} x \d x = 1.
        \end{aligned}
    \end{equation}
    故极限为
    \begin{equation}
        \boxed{\lim_{\alpha \to 0} \int_{-1}^{1} \sqrt{x^2 + \alpha^2} \d x = 1}
    \end{equation}

    \ansitem{2}

    方法一：交换极限与积分顺序。
    设被积函数为 $f(x, \alpha) = \frac{1}{1 + x^2 + \alpha^2}$，其在全平面连续。积分限 $\phi(\alpha) = \alpha$ 与 $\psi(\alpha) = 1 + \alpha$ 在 $\alpha = 0$ 处也连续，由含参量积分的连续性，有
    \begin{equation}
        \begin{aligned}
            \lim_{\alpha \to 0} \int_{\alpha}^{1+\alpha} \frac{1}{1 + x^2 + \alpha^2} \d x & = \int_{0}^{1} \lim_{\alpha \to 0} \frac{1}{1 + x^2 + \alpha^2} \d x \\
                                                                                           & = \int_{0}^{1} \frac{1}{1 + x^2} \d x                                \\
                                                                                           & = \left. \arctan x \right|_{0}^{1} = \frac{\uppi}{4}.
        \end{aligned}
    \end{equation}
    故极限为
    \begin{equation}
        \boxed{\lim_{\alpha \to 0} \int_{\alpha}^{1+\alpha} \frac{1}{1 + x^2 + \alpha^2} \d x = \frac{\uppi}{4}}
    \end{equation}

    方法二：直接积分。
    当 $\alpha \neq 0$ 时，直接积分可得
    \begin{equation}
        \begin{aligned}
            \int_{\alpha}^{1+\alpha} \frac{1}{1 + x^2 + \alpha^2} \d x & = \left[ \frac{1}{\sqrt{1 + \alpha^2}} \arctan \frac{x}{\sqrt{1 + \alpha^2}} \right]_{\alpha}^{1+\alpha}                              \\
                                                                       & = \frac{1}{\sqrt{1 + \alpha^2}} \left( \arctan \frac{1+\alpha}{\sqrt{1+\alpha^2}} - \arctan \frac{\alpha}{\sqrt{1+\alpha^2}} \right).
        \end{aligned}
    \end{equation}
    对其取极限，得
    \begin{equation}
        \begin{aligned}
            \lim_{\alpha \to 0} \int_{\alpha}^{1+\alpha} \frac{1}{1 + x^2 + \alpha^2} \d x & = \lim_{\alpha \to 0} \frac{1}{\sqrt{1 + \alpha^2}} \left( \arctan \frac{1+\alpha}{\sqrt{1+\alpha^2}} - \arctan \frac{\alpha}{\sqrt{1+\alpha^2}} \right) \\
                                                                                           & = 1 \cdot \left( \arctan 1 - \arctan 0 \right) = \frac{\uppi}{4}.
        \end{aligned}
    \end{equation}
    故极限为
    \begin{equation}
        \boxed{\lim_{\alpha \to 0} \int_{\alpha}^{1+\alpha} \frac{1}{1 + x^2 + \alpha^2} \d x = \frac{\uppi}{4}}
    \end{equation}
\end{solution}

\begin{problem}{含参量积分的导数}{Derivative-of-Parametric-Integral}
    求 $F'(\alpha)$：
    \begin{enumerate}
        \item $F(\alpha) = \int_{\sin \alpha}^{\cos \alpha} \e^{\alpha \sqrt{1-x^2}} \d x$;
        \item $F(\alpha) = \int_{a+\alpha}^{b+\alpha} \frac{\sin \alpha x}{x} \d x$;
        \item $F(\alpha) = \int_{0}^{\alpha} \frac{\ln(1 + \alpha x)}{x} \d x$;
        \item $F(\alpha) = \int_{0}^{\alpha} f(x+\alpha, x-\alpha) \d x$（$f(u,v)$ 有连续偏导数）。
    \end{enumerate}
\end{problem}

\begin{solution}
    \ansitem{1}

    设 $\alpha \in \left(-\frac{\uppi}{2}, \frac{\uppi}{2}\right)$。利用含参量积分的求导公式，有
    \begin{equation}
        \begin{aligned}
            F'(\alpha) & = \e^{\alpha \sqrt{1 - \cos^2 \alpha}} \cdot (\cos \alpha)' - \e^{\alpha \sqrt{1 - \sin^2 \alpha}} \cdot (\sin \alpha)' + \int_{\sin \alpha}^{\cos \alpha} \frac{\partial}{\partial \alpha} \left( \e^{\alpha \sqrt{1 - x^2}} \right) \d x \\
                       & = -\sin \alpha \e^{\alpha |\sin \alpha|} - \cos \alpha \e^{\alpha \cos \alpha} + \int_{\sin \alpha}^{\cos \alpha} \sqrt{1-x^2} \e^{\alpha \sqrt{1-x^2}} \d x.
        \end{aligned}
    \end{equation}
    因此有
    \begin{equation}
        \boxed{F'(\alpha) = -\sin \alpha \e^{\alpha |\sin \alpha|} - \cos \alpha \e^{\alpha \cos \alpha} + \int_{\sin \alpha}^{\cos \alpha} \sqrt{1-x^2} \e^{\alpha \sqrt{1-x^2}} \d x}
    \end{equation}

    \ansitem{2}

    当 $\alpha \neq 0$ 时，求导得
    \begin{equation}
        \begin{aligned}
            F'(\alpha) & = \frac{\sin \alpha(b+\alpha)}{b+\alpha} \cdot (b+\alpha)' - \frac{\sin \alpha(a+\alpha)}{a+\alpha} \cdot (a+\alpha)' + \int_{a+\alpha}^{b+\alpha} \frac{\partial}{\partial \alpha} \left( \frac{\sin \alpha x}{x} \right) \d x \\
                       & = \frac{\sin(b\alpha + \alpha^2)}{b+\alpha} - \frac{\sin(a\alpha + \alpha^2)}{a+\alpha} + \int_{a+\alpha}^{b+\alpha} \cos \alpha x \d x                                                                                         \\
                       & = \frac{\sin(b\alpha + \alpha^2)}{b+\alpha} - \frac{\sin(a\alpha + \alpha^2)}{a+\alpha} + \left. \frac{\sin \alpha x}{\alpha} \right|_{a+\alpha}^{b+\alpha}                                                                     \\
                       & = \frac{\sin(b\alpha + \alpha^2)}{b+\alpha} - \frac{\sin(a\alpha + \alpha^2)}{a+\alpha} + \frac{\sin(b\alpha + \alpha^2) - \sin(a\alpha + \alpha^2)}{\alpha}.
        \end{aligned}
    \end{equation}
    因此有
    \begin{equation}
        \boxed{F'(\alpha) = \frac{\sin(b\alpha + \alpha^2)}{b+\alpha} - \frac{\sin(a\alpha + \alpha^2)}{a+\alpha} + \frac{\sin(b\alpha + \alpha^2) - \sin(a\alpha + \alpha^2)}{\alpha}}
    \end{equation}

    \ansitem{3}

    当 $\alpha \neq 0$ 时，求导得
    \begin{equation}
        \begin{aligned}
            F'(\alpha) & = \frac{\ln(1+\alpha^2)}{\alpha} \cdot (\alpha)' - 0 + \int_{0}^{\alpha} \frac{\partial}{\partial \alpha} \left( \frac{\ln(1+\alpha x)}{x} \right) \d x \\
                       & = \frac{\ln(1+\alpha^2)}{\alpha} + \int_{0}^{\alpha} \frac{1}{1 + \alpha x} \d x                                                                        \\
                       & = \frac{\ln(1+\alpha^2)}{\alpha} + \left. \frac{\ln(1+\alpha x)}{\alpha} \right|_{0}^{\alpha}                                                           \\
                       & = \frac{2\ln(1+\alpha^2)}{\alpha}.
        \end{aligned}
    \end{equation}
    因此有
    \begin{equation}
        \boxed{F'(\alpha) = \frac{2\ln(1+\alpha^2)}{\alpha}}
    \end{equation}

    \ansitem{4}

    求导得
    \begin{equation}
        \begin{aligned}
            F'(\alpha) & = f(\alpha+\alpha, \alpha-\alpha) \cdot (\alpha)' - 0 + \int_{0}^{\alpha} \frac{\partial}{\partial \alpha} \left( f(x+\alpha, x-\alpha) \right) \d x \\
                       & = f(2\alpha, 0) + \int_{0}^{\alpha} \left[ f'_1(x+\alpha, x-\alpha) - f'_2(x+\alpha, x-\alpha) \right] \d x.
        \end{aligned}
    \end{equation}
    因此有
    \begin{equation}
        \boxed{F'(\alpha) = f(2\alpha, 0) + \int_{0}^{\alpha} \left[ f'_1(x+\alpha, x-\alpha) - f'_2(x+\alpha, x-\alpha) \right] \d x}
    \end{equation}
\end{solution}

\begin{problem}{变限积分与常微分方程}{ODE-Variable-Limit-Integral}
    设 $f(x)$ 在 $[a, b]$ 上连续，证明：
    \begin{equation}
        y(x) = \frac{1}{k} \int_{c}^{x} f(t) \sin k(x - t) \d t, \quad c, x \in [a, b]
    \end{equation}
    满足常微分方程
    \begin{equation}
        y'' + k^2 y = f(x),
    \end{equation}
    其中 $c$ 与 $k$ 为常数（$k \neq 0$）。
\end{problem}

\begin{myproof}
    利用三角恒等式，将被积函数展开：
    \begin{equation}
        \sin k(x - t) = \sin kx \cos kt - \cos kx \sin kt.
    \end{equation}
    代入 $y(x)$ 的表达式中，由于积分变量为 $t$，可将与 $x$ 相关的项提取至积分号外，得
    \begin{equation}
        y(x) = \frac{1}{k} \sin kx \int_{c}^{x} f(t) \cos kt \d t - \frac{1}{k} \cos kx \int_{c}^{x} f(t) \sin kt \d t.
    \end{equation}
    关于 $x$ 求导，得
    \begin{equation}
        \begin{aligned}
            y'(x) & = \cos kx \int_{c}^{x} f(t) \cos kt \d t + \frac{1}{k} \sin kx \cdot f(x) \cos kx       \\
                  & \quad + \sin kx \int_{c}^{x} f(t) \sin kt \d t - \frac{1}{k} \cos kx \cdot f(x) \sin kx \\
                  & = \cos kx \int_{c}^{x} f(t) \cos kt \d t + \sin kx \int_{c}^{x} f(t) \sin kt \d t.
        \end{aligned}
    \end{equation}
    再次关于 $x$ 求导，得
    \begin{equation}
        \begin{aligned}
            y''(x) & = -k \sin kx \int_{c}^{x} f(t) \cos kt \d t + \cos^2 kx \cdot f(x)                                         \\
                   & \quad + k \cos kx \int_{c}^{x} f(t) \sin kt \d t + \sin^2 kx \cdot f(x)                                    \\
                   & = -k \left( \sin kx \int_{c}^{x} f(t) \cos kt \d t - \cos kx \int_{c}^{x} f(t) \sin kt \d t \right) + f(x) \\
                   & = -k^2 y(x) + f(x).
        \end{aligned}
    \end{equation}
    移项即得
    \begin{equation}
        y'' + k^2 y = f(x).
    \end{equation}
\end{myproof}

\begin{problem}{含参变量积分}{Parameter-dependent-Integrals}
    应用对参数进行微分或积分的方法，计算下列积分：
    \begin{enumerate}
        \item $\int_0^{\frac{\uppi}{2}} \ln(a^2 \sin^2 x + b^2 \cos^2 x) \d x \quad (a > 0, b > 0)$;
        \item $\int_0^{\uppi} \ln(1 - 2a \cos x + a^2) \d x \quad (0 \leqslant a < 1)$;
        \item $\int_0^{\frac{\uppi}{2}} \frac{\arctan(a \tan x)}{\tan x} \d x \quad (a \geqslant 0)$;
        \item $\int_0^{\frac{\uppi}{2}} \ln \frac{1 + a \cos x}{1 - a \cos x} \cdot \frac{\d x}{\cos x} \quad (0 \leqslant a < 1)$.
    \end{enumerate}
\end{problem}

\begin{solution}
    \ansitem{1}

    记所求积分为如下含参变量积分：
    \begin{equation}\label{eq:int1-def}
        I(a, b) = \int_0^{\frac{\uppi}{2}} \ln(a^2 \sin^2 x + b^2 \cos^2 x) \d x.
    \end{equation}
    对 $a$ 求偏导数，得：
    \begin{equation}\label{eq:int1-deriv}
        \begin{aligned}
            \frac{\partial I}{\partial a} & = \int_0^{\frac{\uppi}{2}} \frac{2a \sin^2 x}{a^2 \sin^2 x + b^2 \cos^2 x} \d x \\
                                          & = \int_0^{\frac{\uppi}{2}} \frac{2a \tan^2 x}{a^2 \tan^2 x + b^2} \d x.
        \end{aligned}
    \end{equation}
    令 $t = \tan x$，则 $\d x = \frac{\d t}{1 + t^2}$。代入 \zcref{eq:int1-deriv}，得：
    \begin{equation}
        \begin{aligned}
            \frac{\partial I}{\partial a} & = \int_0^{+\infty} \frac{2a t^2}{(a^2 t^2 + b^2)(1 + t^2)} \d t                                           \\
                                          & = \int_0^{+\infty} \frac{2a}{a^2 - b^2} \left[ \frac{1}{1 + t^2} - \frac{b^2}{a^2 t^2 + b^2} \right] \d t \\
                                          & = \frac{2a}{a^2 - b^2} \left[ \frac{\uppi}{2} - \frac{b}{a} \cdot \frac{\uppi}{2} \right]                 \\
                                          & = \frac{\uppi}{a + b}.
        \end{aligned}
    \end{equation}
    两边对 $a$ 积分，得：
    \begin{equation}
        I(a, b) = \uppi \ln(a + b) + C(b).
    \end{equation}
    由对称性，$I(b, b) = \uppi \ln(2b) + C(b)$。又因为直接积分可得：
    \begin{equation}
        I(b, b) = \int_0^{\frac{\uppi}{2}} \ln(b^2) \d x = \uppi \ln b.
    \end{equation}
    因此 $C(b) = -\uppi \ln 2$。从而有：
    \begin{equation}
        \boxed{I(a, b) = \uppi \ln \frac{a + b}{2}}.
    \end{equation}

    \ansitem{2}

    记含参变量积分为：
    \begin{equation}
        I(a) = \int_0^{\uppi} \ln(1 - 2a \cos x + a^2) \d x.
    \end{equation}
    当 $0 < a < 1$ 时，对 $a$ 求导，得：
    \begin{equation}\label{eq:int2-deriv}
        \begin{aligned}
            I'(a) & = \int_0^{\uppi} \frac{2a - 2\cos x}{1 - 2a \cos x + a^2} \d x                           \\
                  & = \frac{1}{a} \int_0^{\uppi} \left[ 1 + \frac{a^2 - 1}{1 - 2a\cos x + a^2} \right] \d x.
        \end{aligned}
    \end{equation}
    利用万能公式计算积分：
    \begin{equation}
        J = \int_0^{\uppi} \frac{\d x}{1 - 2a \cos x + a^2}.
    \end{equation}
    令 $t = \tan \frac{x}{2}$，则 $\d x = \frac{2\d t}{1 + t^2}$，$\cos x = \frac{1 - t^2}{1 + t^2}$。代入可得：
    \begin{equation}
        \begin{aligned}
            J & = \int_0^{+\infty} \frac{2\d t}{(1 + a^2)(1 + t^2) - 2a(1 - t^2)}                          \\
              & = \int_0^{+\infty} \frac{2}{(1 - a)^2 + (1 + a)^2 t^2} \d t                                \\
              & = \left. \frac{2}{1 - a^2} \arctan\left( \frac{1 + a}{1 - a} t \right) \right|_0^{+\infty} \\
              & = \frac{\uppi}{1 - a^2}.
        \end{aligned}
    \end{equation}
    将 $J$ 的结果代入 \zcref{eq:int2-deriv}，得：
    \begin{equation}
        I'(a) = \frac{\uppi}{a} + \frac{a^2 - 1}{a} \cdot \frac{\uppi}{1 - a^2} = 0.
    \end{equation}
    从而 $I(a)$ 在 $(0, 1)$ 上为常数。由 $I(0) = 0$ 以及积分关于 $a$ 在 $[0, 1)$ 上的连续性，可得：
    \begin{equation}
        \boxed{I(a) = 0}.
    \end{equation}

    \ansitem{3}

    记含参变量积分为：
    \begin{equation}
        I(a) = \int_0^{\frac{\uppi}{2}} \frac{\arctan(a \tan x)}{\tan x} \d x.
    \end{equation}
    对 $a$ 求导，得：
    \begin{equation}\label{eq:int3-deriv}
        I'(a) = \int_0^{\frac{\uppi}{2}} \frac{1}{1 + a^2 \tan^2 x} \d x.
    \end{equation}
    令 $t = \tan x$，则 $\d x = \frac{\d t}{1 + t^2}$。代入 \zcref{eq:int3-deriv}，得：
    \begin{equation}
        \begin{aligned}
            I'(a) & = \int_0^{+\infty} \frac{\d t}{(1 + a^2 t^2)(1 + t^2)}                                               \\
                  & = \frac{1}{1 - a^2} \int_0^{+\infty} \left[ \frac{1}{1 + t^2} - \frac{a^2}{1 + a^2 t^2} \right] \d t \\
                  & = \frac{1}{1 - a^2} \left[ \frac{\uppi}{2} - a \cdot \frac{\uppi}{2} \right]                         \\
                  & = \frac{\uppi}{2(1 + a)}.
        \end{aligned}
    \end{equation}
    两边对 $a$ 积分，并结合 $I(0) = 0$，得：
    \begin{equation}
        \boxed{I(a) = \frac{\uppi}{2} \ln(1 + a)}.
    \end{equation}

    \ansitem{4}

    记含参变量积分为：
    \begin{equation}
        I(a) = \int_0^{\frac{\uppi}{2}} \ln \frac{1 + a \cos x}{1 - a \cos x} \cdot \frac{\d x}{\cos x}.
    \end{equation}
    对 $a$ 求导，得：
    \begin{equation}
        \begin{aligned}
            I'(a) & = \int_0^{\frac{\uppi}{2}} \left( \frac{\cos x}{1 + a \cos x} - \frac{-\cos x}{1 - a \cos x} \right) \frac{\d x}{\cos x} \\
                  & = \int_0^{\frac{\uppi}{2}} \frac{2}{1 - a^2 \cos^2 x} \d x.
        \end{aligned}
    \end{equation}
    分子分母同除以 $\cos^2 x$，得：
    \begin{equation}\label{eq:int4-deriv}
        I'(a) = \int_0^{\frac{\uppi}{2}} \frac{2\sec^2 x}{1 - a^2 + \tan^2 x} \d x.
    \end{equation}
    令 $t = \tan x$，由 \zcref{eq:int4-deriv} 可得：
    \begin{equation}
        \begin{aligned}
            I'(a) & = \int_0^{+\infty} \frac{2}{t^2 + (1 - a^2)} \d t                                                     \\
                  & = \left. \frac{2}{\sqrt{1 - a^2}} \arctan \left( \frac{t}{\sqrt{1 - a^2}} \right) \right|_0^{+\infty} \\
                  & = \frac{\uppi}{\sqrt{1 - a^2}}.
        \end{aligned}
    \end{equation}
    两边对 $a$ 积分，并结合 $I(0) = 0$，得：
    \begin{equation}
        \boxed{I(a) = \uppi \arcsin a}.
    \end{equation}
\end{solution}
